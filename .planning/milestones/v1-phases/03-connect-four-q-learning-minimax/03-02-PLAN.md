Create `src/ConnectFour/Minimax.fs` implementing Negamax Alpha-Beta as documented in RESEARCH.md Pattern 2.

CRITICAL INVARIANTS (from RESEARCH.md common pitfalls):
1. Use `NegInf = -1_000_000` and `PosInf = 1_000_000` as bounds — NOT `System.Int32.MinValue/MaxValue` (overflow when negated)
2. Negate the child score: `childScore = -(minimaxAB nextBoard opp ...)` — this is the negamax invariant; missing this makes the AI try to lose
3. Move ordering: center-first `List.sortBy (fun c -> abs(c - 3))` — reduces node count by 60-80%
4. `pruneCount` is a `ref int` passed through recursion for Alpha-Beta statistics

```fsharp
module ConnectFour.Minimax

open ConnectFour.Domain
open ConnectFour.Rules

// Safe infinity bounds — avoids Int32.MinValue overflow when negated in negamax
let [<Literal>] private NegInf = -1_000_000
let [<Literal>] private PosInf =  1_000_000

/// Score a window of 4 cells from current player's perspective.
/// A window is "alive" only if the opponent has 0 pieces in it (o = 0).
/// Scoring: p=4 → 10000 (win), p=3 → 50, p=2 → 3, p=1 → 1; blocked → 0
let private scoreWindow (window: Cell list) (player: Cell) : int =
    let opp = opponent player
    let p = window |> List.filter ((=) player) |> List.length
    let o = window |> List.filter ((=) opp) |> List.length
    if o > 0 then 0  // blocked window — cannot form 4
    else
        match p with
        | 4 -> 10000
        | 3 -> 50
        | 2 -> 3
        | 1 -> 1
        | _ -> 0

/// Extract all windows of size 4 in all four directions across the board
let private allWindows (board: Board) : Cell list list =
    [ // Horizontal: (rows) × (cols - 3) windows
      for r in 0 .. rows-1 do
        for c in 0 .. cols-4 do
          yield [ board.[idx r c]; board.[idx r (c+1)]; board.[idx r (c+2)]; board.[idx r (c+3)] ]
      // Vertical: (rows - 3) × (cols) windows
      for c in 0 .. cols-1 do
        for r in 0 .. rows-4 do
          yield [ board.[idx r c]; board.[idx (r+1) c]; board.[idx (r+2) c]; board.[idx (r+3) c] ]
      // Diagonal (\): top-left to bottom-right
      for r in 0 .. rows-4 do
        for c in 0 .. cols-4 do
          yield [ board.[idx r c]; board.[idx (r+1) (c+1)]; board.[idx (r+2) (c+2)]; board.[idx (r+3) (c+3)] ]
      // Anti-diagonal (/): bottom-left to top-right
      for r in 3 .. rows-1 do
        for c in 0 .. cols-4 do
          yield [ board.[idx r c]; board.[idx (r-1) (c+1)]; board.[idx (r-2) (c+2)]; board.[idx (r-3) (c+3)] ]
    ]

/// Heuristic board evaluation from current player's perspective.
/// Net score = sum(my windows) - sum(opponent windows) + center column bonus
let evaluateBoard (board: Board) (player: Cell) : int =
    let opp = opponent player
    let windows = allWindows board
    let netWindowScore =
        windows |> List.sumBy (fun w ->
            scoreWindow w player - scoreWindow w opp)
    let centerBonus =
        [ 0 .. rows-1 ]
        |> List.filter (fun r -> board.[idx r 3] = player)
        |> List.length
        |> (*) 3
    netWindowScore + centerBonus

/// Negamax with Alpha-Beta pruning.
/// Returns score from CURRENT PLAYER's perspective.
/// Win/loss scores include depth to prefer faster wins: win → 10000 + depth remaining.
/// pruneCount: mutable counter incremented on each beta cutoff.
let rec minimaxAB
    (board: Board)
    (player: Cell)
    (depth: int)
    (alpha: int)
    (beta: int)
    (pruneCount: int ref) : int =

    match isGameOver board with
    | Some result ->
        match result with
        | RedWins    -> if player = Red    then 10000 + depth else -(10000 + depth)
        | YellowWins -> if player = Yellow then 10000 + depth else -(10000 + depth)
        | Draw       -> 0
    | None when depth = 0 ->
        evaluateBoard board player
    | None ->
        let opp = opponent player
        let moves = legalMoves board |> List.sortBy (fun c -> abs (c - 3))  // center-first
        let rec loop moves alpha bestScore =
            match moves with
            | [] -> bestScore
            | col :: rest ->
                let nextBoard = applyMove board player col
                // NEGAMAX: negate child score (child is evaluated from opponent's perspective)
                let childScore = -(minimaxAB nextBoard opp (depth - 1) (-beta) (-alpha) pruneCount)
                let newBest = max bestScore childScore
                let newAlpha = max alpha newBest
                if newAlpha >= beta then
                    pruneCount.Value <- pruneCount.Value + 1
                    newBest  // Beta cutoff — prune remaining siblings
                else
                    loop rest newAlpha newBest
        loop moves alpha NegInf

/// Choose best column for player using Minimax + Alpha-Beta.
/// Returns (bestColumn, totalPruneCount).
let chooseMoveAB (board: Board) (player: Cell) (depth: int) : int * int =
    let pruneCount = ref 0
    let opp = opponent player
    let moves = legalMoves board |> List.sortBy (fun c -> abs (c - 3))
    let scored =
        moves
        |> List.map (fun col ->
            let nextBoard = applyMove board player col
            let score = -(minimaxAB nextBoard opp (depth - 1) NegInf PosInf pruneCount)
            col, score)
    let bestCol = scored |> List.maxBy snd |> fst
    bestCol, pruneCount.Value

/// Naive full Minimax WITHOUT alpha-beta pruning (for equivalence testing only).
/// Same as minimaxAB but alpha/beta never prune — always explores full tree.
let rec private naiveMinimax
    (board: Board)
    (player: Cell)
    (depth: int) : int =

    match isGameOver board with
    | Some result ->
        match result with
        | RedWins    -> if player = Red    then 10000 + depth else -(10000 + depth)
        | YellowWins -> if player = Yellow then 10000 + depth else -(10000 + depth)
        | Draw       -> 0
    | None when depth = 0 ->
        evaluateBoard board player
    | None ->
        let opp = opponent player
        let moves = legalMoves board |> List.sortBy (fun c -> abs (c - 3))
        moves
        |> List.map (fun col ->
            let nextBoard = applyMove board player col
            -(naiveMinimax nextBoard opp (depth - 1)))
        |> List.max

/// Choose best move using naive full-search Minimax (for test equivalence only).
let chooseMoveNaive (board: Board) (player: Cell) (depth: int) : int =
    let opp = opponent player
    let moves = legalMoves board |> List.sortBy (fun c -> abs (c - 3))
    moves
    |> List.map (fun col ->
        let nextBoard = applyMove board player col
        col, -(naiveMinimax nextBoard opp (depth - 1)))
    |> List.maxBy snd
    |> fst
```

```fsharp
module ConnectFour.Tests.MinimaxTests

open Expecto
open ConnectFour.Domain
open ConnectFour.Rules
open ConnectFour.Minimax

[<Tests>]
let minimaxTests =
    testList "Expecto: Minimax and Alpha-Beta equivalence (CNCT-07)" [

        testCase "Both choose center column (col 3) on empty board at depth 4" <| fun () ->
            let board = emptyBoard ()
            let abMove, pruned = chooseMoveAB board Red 4
            let naiveMove = chooseMoveNaive board Red 4
            Expect.equal abMove 3 "Alpha-Beta should choose center column on empty board"
            Expect.equal naiveMove 3 "Naive Minimax should choose center column on empty board"
            Expect.equal abMove naiveMove "Alpha-Beta and Naive Minimax must agree"
            Expect.isGreaterThan pruned 0 "Alpha-Beta must prune at least one branch"

        testCase "Alpha-Beta agrees with Naive Minimax on same position at depth 4" <| fun () ->
            // A non-trivial position: Red has played col 3, Yellow col 0
            let board = emptyBoard ()
            let board' = applyMove board Red 3
            let board'' = applyMove board' Yellow 0
            let abMove, pruned = chooseMoveAB board'' Red 4
            let naiveMove = chooseMoveNaive board'' Red 4
            Expect.equal abMove naiveMove $"AB chose {abMove}, Naive chose {naiveMove} — must agree"
            Expect.isGreaterThan pruned 0 "Alpha-Beta must prune at least one branch"

        testCase "Minimax finds winning move when 3-in-a-row exists (depth 1)" <| fun () ->
            // Red has 3 in a row horizontally at row 5, cols 0,1,2 — winning move is col 3
            let board = emptyBoard ()
            let board' =
                [0; 1; 2]
                |> List.fold (fun b c ->
                    b |> Array.mapi (fun i cell -> if i = idx 5 c then Red else cell)) board
            let abMove, _ = chooseMoveAB board' Red 1
            Expect.equal abMove 3 "Minimax must find the immediate horizontal win at col 3"

        testCase "Minimax blocks opponent's winning move (depth 2)" <| fun () ->
            // Yellow has 3 in a row horizontally at row 5, cols 0,1,2 — Red must block at col 3
            let board = emptyBoard ()
            let board' =
                [0; 1; 2]
                |> List.fold (fun b c ->
                    b |> Array.mapi (fun i cell -> if i = idx 5 c then Yellow else cell)) board
            let abMove, _ = chooseMoveAB board' Red 2
            Expect.equal abMove 3 "Minimax must block Yellow's imminent horizontal win"

        testCase "chooseMoveAB returns valid column (in legalMoves)" <| fun () ->
            // A partially filled board
            let board = emptyBoard ()
            let board' = applyMove board Red 3
            let board'' = applyMove board' Yellow 2
            let board3 = applyMove board'' Red 4
            let board4 = applyMove board3 Yellow 1
            let abMove, _ = chooseMoveAB board4 Red 5
            let legal = legalMoves board4
            Expect.contains legal abMove $"chooseMoveAB returned {abMove} which is not in legalMoves {legal}"
    ]
```
