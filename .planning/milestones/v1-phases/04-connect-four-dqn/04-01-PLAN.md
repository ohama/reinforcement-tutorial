Write NativeLoader.fs at src/ConnectFourDQN/NativeLoader.fs:

```fsharp
module ConnectFourDQN.NativeLoader

open System.Runtime.InteropServices
open System.IO

// Must be called before ANY TorchSharp type is opened.
// TorchSharp has a known native library discovery bug on macOS ARM64:
// it derives packagesDir from the build output's parent directory, which is wrong.
// Solution: load native libs explicitly from AppContext.BaseDirectory/runtimes/osx-arm64/native/
// Load order matters: libomp -> libc10 -> libtorch_cpu -> libtorch -> LibTorchSharp
let private load () =
    let exeDir = System.AppContext.BaseDirectory
    let nativeDir = Path.Combine(exeDir, "runtimes", "osx-arm64", "native")
    if Directory.Exists(nativeDir) then
        NativeLibrary.Load(Path.Combine(nativeDir, "libomp.dylib"))           |> ignore
        NativeLibrary.Load(Path.Combine(nativeDir, "libc10.dylib"))           |> ignore
        NativeLibrary.Load(Path.Combine(nativeDir, "libtorch_cpu.dylib"))     |> ignore
        NativeLibrary.Load(Path.Combine(nativeDir, "libtorch.dylib"))         |> ignore
        NativeLibrary.Load(Path.Combine(nativeDir, "libLibTorchSharp.dylib")) |> ignore
    // On Linux/Windows, runtimes dir is absent; TorchSharp auto-discovery works there

do load ()
```

Replace the DQNAgent.fs stub with the full boardToTensor implementation:

```fsharp
module ConnectFourDQN.DQNAgent

open TorchSharp
open type TorchSharp.torch
open ConnectFourDQN.Domain

// Encode a Connect Four board as a 3-channel float32 tensor.
// Shape: [3, 6, 7]
//   Channel 0: 1.0 where board has myPiece, else 0.0
//   Channel 1: 1.0 where board has oppPiece, else 0.0
//   Channel 2: 1.0 where board is Empty, else 0.0
// Invariant (DQN-08): sum of all values = 6 * 7 = 42.0 for any valid board
//   (exactly one channel = 1.0 per cell)
let boardToTensor (board: Board) (myPiece: Cell) (oppPiece: Cell) : Tensor =
    let flat = Array.init (3 * 6 * 7) (fun i ->
        let ch   = i / (6 * 7)
        let rem  = i % (6 * 7)
        let cell = board.[rem / 7 * 7 + rem % 7]
        match ch, cell with
        | 0, c when c = myPiece  -> 1.0f
        | 1, c when c = oppPiece -> 1.0f
        | 2, Empty               -> 1.0f
        | _                      -> 0.0f)
    torch.tensor(flat, dtype=ScalarType.Float32).reshape([| 3L; 6L; 7L |])
```

```fsharp
module ConnectFourDQN.Tests.TensorTests

open Expecto
open FsCheck
open ConnectFourDQN.Domain
open ConnectFourDQN.Rules
open ConnectFourDQN.DQNAgent

// FsCheck generator: produce a valid board by playing N random plies
let genValidBoard =
    Gen.choose (0, 30) |> Gen.bind (fun plies ->
        let rng = System.Random(42)
        let mutable board = Array.create (rows * cols) Empty
        let mutable current = Red
        let mutable remaining = plies
        while remaining > 0 do
            let legal = legalMoves board
            if not (List.isEmpty legal) && not (isGameOver board) then
                let col = legal.[rng.Next(legal.Length)]
                board <- applyMove board current col
                current <- if current = Red then Yellow else Red
            remaining <- remaining - 1
        board)

type ValidBoard = ValidBoard of Board

type Arb =
    static member ValidBoard() =
        { new Arbitrary<ValidBoard>() with
            override _.Generator = genValidBoard |> Gen.map ValidBoard
            override _.Shrinker _ = Seq.empty }

[<Tests>]
let tensorSumInvariant =
    testProperty "boardToTensor: sum of all channels = 42.0 for any valid board" (
        Prop.forAll (Arb.ValidBoard()) (fun (ValidBoard board) ->
            // DQN-11: use NewDisposeScope to ensure no tensor leak during test
            use _scope = TorchSharp.torch.NewDisposeScope()
            use t = boardToTensor board Red Yellow
            let sum = t.sum().item<float32>()
            abs(sum - 42.0f) < 0.001f))

[<Tests>]
let tensorShapeTest =
    testCase "boardToTensor: output shape is [3, 6, 7]" (fun () ->
        use _scope = TorchSharp.torch.NewDisposeScope()
        let board = Array.create (rows * cols) Empty
        use t = boardToTensor board Red Yellow
        Expect.equal (t.shape) [| 3L; 6L; 7L |] "Shape must be [3,6,7]")

[<Tests>]
let emptyBoardAllEmpty =
    testCase "boardToTensor: empty board has channel 2 = all 1.0" (fun () ->
        use _scope = TorchSharp.torch.NewDisposeScope()
        let board = Array.create (rows * cols) Empty
        use t = boardToTensor board Red Yellow
        // Channel 2 (empty) should sum to 42.0
        use ch2 = t.[2L]
        let ch2sum = ch2.sum().item<float32>()
        Expect.isTrue (abs(ch2sum - 42.0f) < 0.001f) "Empty board: channel 2 sum = 42")
```
