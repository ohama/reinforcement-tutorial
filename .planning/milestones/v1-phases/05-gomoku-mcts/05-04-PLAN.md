Key patterns:
- TrainingSample stores float32[] not Tensor — tensors outside dispose scope leak memory in long loops
- SelfPlay value target convention: after game ends, winner's states get +1.0f, loser's get -1.0f, draw = 0.0f
  Note: each state is stored from the perspective of the player who was to move at that state
  → if state.CurrentPlayer was Black and Black won, valueTarget = +1.0f; if Black lost, valueTarget = -1.0f
- Temperature schedule: τ=1.0 for first 15 moves (sample from visit distribution), τ→0 after (argmax)
- model.eval() before self-play, model.train() during trainBatch, model.eval() after
- torch.no_grad() NOT needed in trainBatch — keep it simple; training tensors don't leak due to NewDisposeScope()
- Policy cross-entropy loss: -(policyT * logProbs).sum(1L).mean()
- Value MSE loss: functional.mse_loss(valuePred, valueT, Reduction.Mean)
- Adam optimizer, lr=2e-3 (from junxiaosong reference)
- Program.fs is sole impure file — Training.fs is pure (returns data/metrics only)
- isGameOver equivalent: match state.LastMove with | Some m when isWinningMove state.Board m → terminal
- Serilog logs added in Plan 05; Program.fs in this plan uses printfn for logging (to be enhanced later)
- Human vs AI: player enters row,col (0-indexed) or flat index; AI runs mctsSearchWithNet (GMOK-07)
- Model save: model.save("gomoku_model.pt"); load: model.load("gomoku_model.pt") — from Phase 4 pattern
- open type TorchSharp.torch shadows int/int64 — use Operators.int / Operators.int64

module Gomoku.SelfPlay

open TorchSharp
open type TorchSharp.torch
open Gomoku.Domain
open Gomoku.Rules
open Gomoku.PolicyValueNet
open Gomoku.Mcts

/// Training sample: stored as float32 arrays (NOT Tensors) to avoid memory leaks in replay buffer.
type TrainingSample = {
    State:        float32[]  // flat [4 * 15 * 15] = 900 values, from boardToTensor
    PolicyTarget: float32[]  // flat [225] MCTS visit distribution (sums to 1)
    ValueTarget:  float32    // +1.0f = current player won, -1.0f = lost, 0.0f = draw
}

/// Temperature-based move selection from MCTS visit counts.
/// τ=1.0: sample proportionally to visit counts (first 15 moves)
/// τ→0: take argmax (greedy, after move 15)
let private selectMoveByTemp (rng: System.Random) (visitProbs: (int * float) array) (moveCount: int) : int =
    if moveCount >= 15 then
        // Greedy: highest visit count
        visitProbs |> Array.maxBy snd |> fst
    else
        // Sample proportionally to visit counts
        let weights = visitProbs |> Array.map snd
        let total   = Array.sum weights
        let r = rng.NextDouble() * total
        let mutable cumSum = 0.0
        let mutable chosen = fst visitProbs.[0]
        let mutable found  = false
        for (action, w) in visitProbs do
            if not found then
                cumSum <- cumSum + w
                if cumSum >= r then
                    chosen <- action
                    found  <- true
        chosen

/// Play one complete self-play game and return training samples with value targets filled.
/// model should be in eval() mode before calling this function.
/// nSimulations: MCTS simulations per move (100 for training, 400 for evaluation)
let playSelfPlayGame
    (model: PolicyValueNet)
    (rng: System.Random)
    (nSimulations: int)
    (cPuct: float)
    : TrainingSample[] =

    let mutable state = initialState ()
    // Accumulate (state_tensor_data, policy_pi, whoseTurn) — value target filled after game
    let stateHistory  = System.Collections.Generic.List<float32[]>()   // board encoding
    let policyHistory = System.Collections.Generic.List<float32[]>()   // MCTS visit dist
    let playerHistory = System.Collections.Generic.List<Player>()      // current player at each state

    let mutable running = true

    while running do
        // Capture state encoding BEFORE making the move (current player's perspective)
        use _encScope = torch.NewDisposeScope()
        let stateTensor = boardToTensor state
        let stateData   = stateTensor.data<float32>().ToArray()
        stateHistory.Add(stateData)
        playerHistory.Add(state.CurrentPlayer)

        // MCTS search with Dirichlet noise (addNoise=true during training)
        let visitProbs = mctsSearchWithNet model state nSimulations cPuct true

        // Build policy target from visit counts
        let piTarget = Array.zeroCreate<float32> (BoardSize * BoardSize)
        for (action, prob) in visitProbs do
            piTarget.[action] <- float32 prob
        policyHistory.Add(piTarget)

        // Select move
        let move = selectMoveByTemp rng visitProbs state.MoveCount
        state <- applyMove state move

        // Check terminal
        match state.LastMove with
        | Some m when isWinningMove state.Board m -> running <- false
        | _ when (legalMoves state.Board).Length = 0 -> running <- false
        | _ -> ()

    // Determine game outcome
    // state.CurrentPlayer is the player who was SUPPOSED to move but game ended
    // The winner is the player who made the last move = opponent of state.CurrentPlayer
    let gameOutcome =
        match state.LastMove with
        | Some m when isWinningMove state.Board m ->
            Some (opponent state.CurrentPlayer)  // winner
        | _ -> None  // draw

    // Fill value targets: +1.0 if that player won, -1.0 if lost, 0.0 if draw
    let n = stateHistory.Count
    Array.init n (fun i ->
        let valueTarget =
            match gameOutcome with
            | Some winner ->
                if playerHistory.[i] = winner then 1.0f else -1.0f
            | None -> 0.0f
        { State        = stateHistory.[i]
          PolicyTarget = policyHistory.[i]
          ValueTarget  = valueTarget })
```

module Gomoku.Training

open TorchSharp
open type TorchSharp.torch
open type TorchSharp.torch.nn
open type TorchSharp.torch.optim
open Gomoku.Domain
open Gomoku.PolicyValueNet
open Gomoku.SelfPlay

/// Training configuration record.
type TrainingConfig = {
    NSelfPlayGames:   int     // Self-play games per iteration (default 1-5)
    NSimulations:     int     // MCTS simulations per move during self-play (default 100)
    BatchSize:        int     // Training batch size (default 256)
    MaxBufferSize:    int     // Replay buffer size in samples (default 10000)
    LearningRate:     float   // Adam learning rate (default 2e-3)
    CPuct:            float   // PUCT exploration constant (default 5.0)
    NTrainingIter:    int     // Total training iterations (default 200)
    NEpochsPerIter:   int     // Gradient steps per iteration (default 5)
    ModelSavePath:    string  // Path to save model (default "gomoku_model.pt")
}

let defaultConfig = {
    NSelfPlayGames  = 1
    NSimulations    = 100
    BatchSize       = 256
    MaxBufferSize   = 10000
    LearningRate    = 2e-3
    CPuct           = 5.0
    NTrainingIter   = 200
    NEpochsPerIter  = 5
    ModelSavePath   = "gomoku_model.pt"
}

/// Single gradient update on one batch.
/// model must be in train() mode. Returns (policyLoss, valueLoss) as floats.
/// PURE — no I/O.
let trainBatch (model: PolicyValueNet) (opt: Optimizer) (batch: TrainingSample[]) : float * float =
    use _scope = torch.NewDisposeScope()
    // Stack states [B, 4, 15, 15]
    let stateFlat  = batch |> Array.collect (fun s -> s.State)
    let stateT     = torch.tensor(stateFlat, dtype=ScalarType.Float32)
                           .reshape([| Operators.int64 batch.Length; 4L;
                                       Operators.int64 BoardSize; Operators.int64 BoardSize |])
    // Policy targets [B, 225]
    let policyFlat = batch |> Array.collect (fun s -> s.PolicyTarget)
    let policyT    = torch.tensor(policyFlat, dtype=ScalarType.Float32)
                           .reshape([| Operators.int64 batch.Length;
                                       Operators.int64 (BoardSize * BoardSize) |])
    // Value targets [B, 1]
    let valueFlat  = batch |> Array.map (fun s -> s.ValueTarget)
    let valueT     = torch.tensor(valueFlat, dtype=ScalarType.Float32).unsqueeze(1L)

    // Forward pass — model should already be in train() mode
    let (logProbs, valuePred) = model.forwardBoth(stateT)

    // Policy cross-entropy loss: -sum(pi * log_p) per sample, averaged over batch
    let policyLoss = -(policyT * logProbs).sum(1L).mean()

    // Value MSE loss
    let valueLoss = functional.mse_loss(valuePred, valueT, Reduction.Mean)

    let loss = policyLoss + valueLoss
    opt.zero_grad()
    loss.backward()
    opt.step() |> ignore

    (Operators.float (policyLoss.item<float32>()),
     Operators.float (valueLoss.item<float32>()))

/// Training result for each iteration.
type IterationResult = {
    Iteration:   int
    GamesPlayed: int
    AvgPolicyLoss: float
    AvgValueLoss:  float
}

/// Full self-play training pipeline.
/// Pure function — returns IterationResult list. All I/O (logging) done in Program.fs.
let runSelfPlayTraining
    (model: PolicyValueNet)
    (config: TrainingConfig)
    (rng: System.Random)
    : IterationResult list =

    let opt = Adam(model.parameters(), config.LearningRate)
    let buffer = System.Collections.Generic.List<TrainingSample>()
    let results = System.Collections.Generic.List<IterationResult>()

    for iter in 1 .. config.NTrainingIter do
        // 1. Self-play: collect data
        model.eval()
        for _ in 1 .. config.NSelfPlayGames do
            let samples = playSelfPlayGame model rng config.NSimulations config.CPuct
            buffer.AddRange(samples)
            // Trim buffer to max size (keep newest)
            while buffer.Count > config.MaxBufferSize do
                buffer.RemoveAt(0)

        // 2. Training: batch gradient updates
        model.train()
        let mutable totalPolicyLoss = 0.0
        let mutable totalValueLoss  = 0.0
        let mutable nBatches = 0

        if buffer.Count >= config.BatchSize then
            for _ in 1 .. config.NEpochsPerIter do
                // Sample random batch
                let batchIndices =
                    Array.init config.BatchSize (fun _ -> rng.Next(buffer.Count))
                let batch =
                    batchIndices |> Array.map (fun i -> buffer.[i])
                let (pLoss, vLoss) = trainBatch model opt batch
                totalPolicyLoss <- totalPolicyLoss + pLoss
                totalValueLoss  <- totalValueLoss  + vLoss
                nBatches        <- nBatches + 1

        model.eval()

        let result = {
            Iteration      = iter
            GamesPlayed    = iter * config.NSelfPlayGames
            AvgPolicyLoss  = if nBatches > 0 then totalPolicyLoss / float nBatches else 0.0
            AvgValueLoss   = if nBatches > 0 then totalValueLoss  / float nBatches else 0.0
        }
        results.Add(result)

    results |> Seq.toList

/// Evaluate model win rate vs random opponent (pure MCTS, no network).
/// Returns win count out of nGames.
let evaluateVsRandom
    (model: PolicyValueNet)
    (nGames: int)
    (nSimulations: int)
    (cPuct: float)
    (rng: System.Random)
    : int =
    let mutable wins = 0
    model.eval()
    for i in 1 .. nGames do
        let mctsIsBlack = (i % 2 = 1)
        let mctsPlayer  = if mctsIsBlack then Domain.Black else Domain.White
        let mutable state  = Domain.initialState ()
        let mutable running = true
        while running do
            match state.LastMove with
            | Some m when Rules.isWinningMove state.Board m ->
                let winner = Domain.opponent state.CurrentPlayer
                if winner = mctsPlayer then wins <- wins + 1
                running <- false
            | _ when (Rules.legalMoves state.Board).Length = 0 ->
                running <- false
            | _ ->
                let legal = Rules.legalMoves state.Board
                let move =
                    if state.CurrentPlayer = mctsPlayer then
                        let vp = Mcts.mctsSearchWithNet model state nSimulations cPuct false
                        Mcts.bestMove vp
                    else
                        legal.[rng.Next(legal.Length)]
                state <- Rules.applyMove state move
    wins

module Gomoku.Console.Program

open System
open Gomoku.Domain
open Gomoku.Rules
open Gomoku.PolicyValueNet
open Gomoku.Training

// ── Board Display ─────────────────────────────────────────────────────────────

let private printBoard (board: Board) =
    printfn ""
    printf "   "
    for c in 0 .. BoardSize - 1 do printf "%2d" c
    printfn ""
    for r in 0 .. BoardSize - 1 do
        printf "%2d " r
        for c in 0 .. BoardSize - 1 do
            let cell = board.[r * BoardSize + c]
            let ch = if cell = 1 then "●" elif cell = -1 then "○" else "·"
            printf " %s" ch
        printfn ""
    printfn ""

// ── Human Move Input ──────────────────────────────────────────────────────────

/// Parse human input: "row,col" or flat index.
let private parseHumanMove (input: string) (legalMoves: int[]) : int option =
    let parts = input.Trim().Split([| ','; ' ' |], StringSplitOptions.RemoveEmptyEntries)
    match parts with
    | [| r; c |] ->
        match Int32.TryParse(r), Int32.TryParse(c) with
        | (true, row), (true, col) when row >= 0 && row < BoardSize && col >= 0 && col < BoardSize ->
            let idx = row * BoardSize + col
            if legalMoves |> Array.contains idx then Some idx else None
        | _ -> None
    | [| idx |] ->
        match Int32.TryParse(idx) with
        | true, i when i >= 0 && i < BoardSize * BoardSize && legalMoves |> Array.contains i -> Some i
        | _ -> None
    | _ -> None

// ── Game Loop ─────────────────────────────────────────────────────────────────

/// Training mode: run self-play training loop with printfn progress.
/// Returns trained model (saves to disk if ModelSavePath set).
let private runTraining (config: TrainingConfig) =
    printfn "\n=== Gomoku MCTS Training ==="
    printfn "Config: %d iterations, %d simulations/move, %d games/iter"
        config.NTrainingIter config.NSimulations config.NSelfPlayGames
    printfn "Press Ctrl+C to stop early (model saves on exit).\n"

    let model = new PolicyValueNet("pv-net")
    let rng   = Random(42)

    // Training loop (Serilog wired in Plan 05; using printfn here)
    let results = runSelfPlayTraining model config rng

    // Summary
    printfn "\n=== Training Complete ==="
    for r in results |> List.filter (fun r -> r.Iteration % 10 = 0 || r.Iteration = 1) do
        printfn "  Iter %3d | Games=%4d | PolicyLoss=%.4f | ValueLoss=%.4f"
            r.Iteration r.GamesPlayed r.AvgPolicyLoss r.AvgValueLoss

    // Save model
    model.save(config.ModelSavePath) |> ignore
    printfn "\nModel saved to: %s" config.ModelSavePath
    model

/// Benchmark mode: load model, evaluate vs random opponent.
let private runBenchmark (modelPath: string) (nGames: int) (nSimulations: int) =
    printfn "\n=== Benchmark: AI vs Random (%d games, %d simulations/move) ==="
        nGames nSimulations

    let model = new PolicyValueNet("pv-net")
    if IO.File.Exists(modelPath) then
        model.load(modelPath) |> ignore
        printfn "Model loaded from: %s" modelPath
    else
        printfn "No saved model found at %s — using random weights." modelPath

    let rng  = Random(123)
    let wins = evaluateVsRandom model nGames nSimulations 5.0 rng
    let rate = float wins / float nGames
    printfn "Result: %d/%d wins (%.1f%%)" wins nGames (rate * 100.0)
    if rate > 0.80 then printfn "✓ Win rate >80%% vs random!"
    else printfn "✗ Win rate <=80%% vs random."

/// Human vs AI mode (GMOK-07).
let private runHumanVsAI (modelPath: string) (nSimulations: int) =
    printfn "\n=== Human vs AI (difficulty = %d simulations/move) ===" nSimulations

    let model = new PolicyValueNet("pv-net")
    if IO.File.Exists(modelPath) then
        model.load(modelPath) |> ignore
        printfn "Model loaded from: %s\n" modelPath
    else
        printfn "No saved model found at %s — AI uses random weights.\n" modelPath

    printfn "You are ○ (White). AI is ● (Black)."
    printfn "Enter moves as 'row,col' (0-indexed) or flat index 0-224.\n"

    let rng = Random()
    let mutable state  = initialState ()
    let mutable running = true
    let humanPlayer = White

    while running do
        printBoard state.Board

        // Check terminal before asking for move
        match state.LastMove with
        | Some m when isWinningMove state.Board m ->
            let winner = opponent state.CurrentPlayer
            if winner = humanPlayer then printfn "You win!"
            else printfn "AI wins!"
            running <- false
        | _ when (legalMoves state.Board).Length = 0 ->
            printfn "Draw!"
            running <- false
        | _ ->
            if state.CurrentPlayer = humanPlayer then
                // Human move
                printf "Your move: "
                let input = Console.ReadLine()
                let legal = legalMoves state.Board
                match parseHumanMove input legal with
                | Some move -> state <- applyMove state move
                | None      -> printfn "Invalid move. Enter 'row,col' or flat index in [0,224]."
            else
                // AI move
                printfn "AI thinking..."
                let visitProbs = Mcts.mctsSearchWithNet model state nSimulations 5.0 false
                let move = Mcts.bestMove visitProbs
                let r = move / BoardSize
                let c = move % BoardSize
                printfn "AI plays: (%d, %d)" r c
                state <- applyMove state move

// ── Menu ──────────────────────────────────────────────────────────────────────

let private showMenu () =
    printfn "\n=== Gomoku MCTS ==="
    printfn "  1. Train (self-play)"
    printfn "  2. Benchmark vs random"
    printfn "  3. Human vs AI"
    printfn "  4. Quit"
    printf "Choice: "

[<EntryPoint>]
let main _ =
    let modelPath    = "gomoku_model.pt"
    let trainConfig  = { defaultConfig with ModelSavePath = modelPath }

    let mutable running = true
    while running do
        showMenu ()
        match Console.ReadLine() with
        | "1" -> runTraining trainConfig |> ignore
        | "2" ->
            printf "Simulations per move (default 100): "
            let sims =
                match Int32.TryParse(Console.ReadLine()) with
                | true, n when n > 0 -> n
                | _ -> 100
            runBenchmark modelPath 50 sims
        | "3" ->
            printf "Simulations per move / difficulty (default 100): "
            let sims =
                match Int32.TryParse(Console.ReadLine()) with
                | true, n when n > 0 -> n
                | _ -> 100
            runHumanVsAI modelPath sims
        | "4" | "q" | "Q" -> running <- false
        | other ->
            if other <> "" then printfn "Unknown option: %s" other

    0
```
