```fsharp
module TicTacToe.Domain

type Cell = Empty | X | O

/// Flat 9-element board (0=top-left, 8=bottom-right, row-major)
/// 0|1|2
/// 3|4|5
/// 6|7|8
type Board = Cell array

/// Immutable game state (TICT-02)
type GameState = {
    Board: Board
    CurrentPlayer: Cell  // X 또는 O; Empty는 절대 불가
}

/// Value table: board state -> X의 예상 승률 [0.0, 1.0]
type ValueTable = Map<Board, float>

/// 9개 빈 칸으로 초기화된 보드 생성
/// CRITICAL: Array.zeroCreate 사용 금지 — DU 타입에 null 반환함
/// 반드시 Array.create 9 Empty 사용
let emptyBoard () : Board = Array.create 9 Empty

let initialState () : GameState =
    { Board = emptyBoard (); CurrentPlayer = X }

let otherPlayer (p: Cell) : Cell =
    match p with
    | X -> O
    | O -> X
    | Empty -> failwith "otherPlayer: Empty는 유효한 플레이어가 아닙니다"
```

**Step 4 — Rules.fs 작성** (`src/TicTacToe/Rules.fs`):
```fsharp
module TicTacToe.Rules

open TicTacToe.Domain

/// 8개의 승리 라인 (행 3개 + 열 3개 + 대각선 2개)
let private winLines = [|
    [|0;1;2|]; [|3;4;5|]; [|6;7;8|]  // 행
    [|0;3;6|]; [|1;4;7|]; [|2;5;8|]  // 열
    [|0;4;8|]; [|2;4;6|]             // 대각선
|]

/// 승자 확인: 승자가 있으면 Some player, 없으면 None
let checkWinner (board: Board) : Cell option =
    winLines
    |> Array.tryPick (fun line ->
        let cells = line |> Array.map (fun i -> board.[i])
        if cells.[0] <> Empty && cells.[0] = cells.[1] && cells.[1] = cells.[2]
        then Some cells.[0]
        else None)

type GameResult = XWins | OWins | Draw

/// 게임 종료 여부 확인: 종료면 Some GameResult, 진행 중이면 None
let isGameOver (board: Board) : GameResult option =
    match checkWinner board with
    | Some X -> Some XWins
    | Some O -> Some OWins
    | Some Empty -> None  // 논리적으로 불가능하나 패턴 매칭 완전성 충족
    | None ->
        if Array.forall (fun c -> c <> Empty) board
        then Some Draw
        else None

/// 합법적인 수 목록 (Empty 칸의 인덱스 목록)
let legalMoves (board: Board) : int list =
    board
    |> Array.indexed
    |> Array.choose (fun (i, c) -> if c = Empty then Some i else None)
    |> Array.toList

/// 이동 적용: 해당 인덱스를 player로 설정한 새 보드 반환
/// 합법성 검사는 호출자 책임
let applyMove (board: Board) (player: Cell) (index: int) : Board =
    board |> Array.mapi (fun i c -> if i = index then player else c)
```

**Step 3 — PropertyTests.fs 작성** (`tests/TicTacToe.Tests/PropertyTests.fs`):

TICT-07 요구사항: 빈칸 수 감소, 차례 교대, 합법 수 범위, applyMove 단일 칸 변경, 초기 보드 9칸 등의 불변 조건.

```fsharp
module TicTacToe.Tests.PropertyTests

open Expecto
open Expecto.ExpectoFsCheck
open TicTacToe.Domain
open TicTacToe.Rules

[<Tests>]
let propertyTests =
    testList "FsCheck 보드 불변 조건" [

        // TICT-07: 이동 후 빈칸 수가 1 감소한다
        testProperty "빈칸 수는 합법적인 이동 1회 후 1 감소한다" <| fun () ->
            let board = Array.create 9 Empty
            let emptiesBefore = board |> Array.filter ((=) Empty) |> Array.length
            let board' = applyMove board X 4  // 중앙 — 빈 보드에서 항상 합법
            let emptiesAfter = board' |> Array.filter ((=) Empty) |> Array.length
            emptiesAfter = emptiesBefore - 1

        // TICT-07: X가 먼저 두면 다음은 O 차례
        testProperty "플레이어는 교대한다: X 이후 O 차례" <| fun () ->
            let state = initialState ()
            let nextPlayer = otherPlayer state.CurrentPlayer
            nextPlayer = O  // X로 시작, 다음은 O

        // TICT-07: 합법 수는 모두 0-8 범위 내
        testProperty "합법 수는 모두 [0, 8] 범위 내에 있다" <| fun () ->
            let board = Array.create 9 Empty
            let moves = legalMoves board
            moves |> List.forall (fun m -> m >= 0 && m <= 8)

        // TICT-07: applyMove는 대상 칸만 변경한다
        testProperty "applyMove는 대상 칸만 변경하고 나머지는 유지한다" <| fun () ->
            let board = Array.create 9 Empty
            let board' = applyMove board X 4
            let unchanged = [0;1;2;3;5;6;7;8] |> List.forall (fun i -> board'.[i] = Empty)
            unchanged && board'.[4] = X

        // TICT-07: 초기 보드는 9개 빈 칸
        testProperty "초기 보드는 9개의 빈 칸을 가진다" <| fun () ->
            let board = emptyBoard ()
            board |> Array.filter ((=) Empty) |> Array.length = 9

    ]
```
