Create `src/ConnectFour/QAgent.fs` implementing tabular Q-Learning as documented in RESEARCH.md Pattern 3.

CRITICAL DESIGN DECISIONS (from RESEARCH.md):
1. Use SEPARATE Q-tables for Red and Yellow (avoid perspective confusion — RESEARCH.md Pitfall 2)
2. String key for Q-table: `encodeState` returns a 42-char string ('R', 'Y', '.') — debuggable and fast
3. Q-values are `float[]` length 7, mutable in-place (Dictionary value is reference type, so `Array.set` works without re-inserting)
4. `chooseAction` must ONLY select among legal columns (from `legalMoves`) — never play into a full column
5. Reward structure: win=+1.0, loss=-1.0, draw=+0.3, intermediate=0.0 (sparse rewards)

```fsharp
module ConnectFour.QAgent

open ConnectFour.Domain
open ConnectFour.Rules

/// Q-table: board state string key → Q-values per column (float array of length 7)
type QTable = System.Collections.Generic.Dictionary<string, float[]>

/// Encode board as 42-character string: 'R'=Red, 'Y'=Yellow, '.'=Empty
let encodeState (board: Board) : string =
    board
    |> Array.map (fun c ->
        match c with
        | Red    -> 'R'
        | Yellow -> 'Y'
        | Empty  -> '.')
    |> System.String

/// Create a new, empty Q-table
let createQTable () : QTable = QTable()

/// Get Q-values for a state; initializes to 0.0 if unseen
let getQ (table: QTable) (state: string) : float[] =
    match table.TryGetValue(state) with
    | true, values -> values
    | false, _ ->
        let values = Array.create cols 0.0
        table.[state] <- values
        values

/// Choose action using epsilon-greedy policy.
/// Returns a legal column (guaranteed to be in legalMoves board).
/// Illegal columns are never selected (Q-values for illegal columns are irrelevant).
let chooseAction (rng: System.Random) (table: QTable) (board: Board) (epsilon: float) : int =
    let legal = legalMoves board
    if legal.IsEmpty then failwith "chooseAction: no legal moves (game should be over)"
    if rng.NextDouble() < epsilon then
        legal.[rng.Next(legal.Length)]
    else
        let state = encodeState board
        let qVals = getQ table state
        legal |> List.maxBy (fun c -> qVals.[c])

/// Q-learning update: Q(s,a) ← Q(s,a) + α × [r + γ × maxQ(s') − Q(s,a)]
/// For terminal states, the future value is 0.0 (isTerminal = true).
let updateQ
    (table: QTable)
    (state: string)
    (action: int)
    (reward: float)
    (nextState: string)
    (nextLegalMoves: int list)
    (alpha: float)
    (gamma: float)
    (isTerminal: bool) : unit =

    let qCurr = getQ table state
    let nextMax =
        if isTerminal || nextLegalMoves.IsEmpty then 0.0
        else
            let qNext = getQ table nextState
            nextLegalMoves |> List.map (fun c -> qNext.[c]) |> List.max
    let target = reward + gamma * nextMax
    qCurr.[action] <- qCurr.[action] + alpha * (target - qCurr.[action])
    // No re-insert needed: Dictionary stores reference to the float[] array

/// Reward constants
let [<Literal>] RewardWin  =  1.0
let [<Literal>] RewardLoss = -1.0
let [<Literal>] RewardDraw =  0.3
let [<Literal>] RewardStep =  0.0
```

```fsharp
module ConnectFour.Console.Training

open ConnectFour.Domain
open ConnectFour.Rules
open ConnectFour.QAgent

/// Result of a single training episode
type EpisodeOutcome = RedWon | YellowWon | Drew

/// Result returned after a full training run
type TrainingResult = {
    RedTable:       QTable
    YellowTable:    QTable
    TotalEpisodes:  int
    /// Snapshot of (episode, qTableSize, redWinRate) every logInterval episodes
    History:        (int * int * float) list
}

/// Play one self-play episode.
/// Both Red and Yellow use their respective Q-tables with epsilon-greedy action selection.
/// Returns (redTable', yellowTable', outcome) — tables are mutated in-place.
let playEpisode
    (rng: System.Random)
    (redTable: QTable)
    (yellowTable: QTable)
    (alpha: float)
    (gamma: float)
    (epsilon: float) : EpisodeOutcome =

    let rec loop (board: Board) (currentPlayer: Cell) =
        match isGameOver board with
        | Some RedWins    -> RedWon
        | Some YellowWins -> YellowWon
        | Some Draw       -> Drew
        | None ->
            let table   = if currentPlayer = Red then redTable else yellowTable
            let oppTable = if currentPlayer = Red then yellowTable else redTable
            let state   = encodeState board
            let action  = chooseAction rng table board epsilon
            let nextBoard = applyMove board currentPlayer action

            match isGameOver nextBoard with
            | Some RedWins ->
                // Current player won
                let reward = if currentPlayer = Red then RewardWin else RewardLoss
                updateQ table state action reward (encodeState nextBoard) [] alpha gamma true
                // Update opponent's last-state with loss (handled next episode — simplified approach)
                if currentPlayer = Red then RedWon else YellowWon
            | Some YellowWins ->
                let reward = if currentPlayer = Yellow then RewardWin else RewardLoss
                updateQ table state action reward (encodeState nextBoard) [] alpha gamma true
                if currentPlayer = Yellow then YellowWon else RedWon
            | Some Draw ->
                updateQ table state action RewardDraw (encodeState nextBoard) [] alpha gamma true
                Drew
            | None ->
                // Non-terminal move: reward = 0, update with next state
                let nextState = encodeState nextBoard
                let nextLegal = legalMoves nextBoard
                updateQ table state action RewardStep nextState nextLegal alpha gamma false
                let nextPlayer = opponent currentPlayer
                loop nextBoard nextPlayer

    loop (emptyBoard ()) Red

/// Train Q-Learning agents via self-play for `episodes` episodes.
/// Logs a snapshot every `logInterval` episodes.
/// Returns TrainingResult with both tables and history.
let trainQLearning
    (rng: System.Random)
    (episodes: int)
    (alpha: float)
    (gamma: float)
    (epsilonStart: float)
    (epsilonEnd: float)
    (logInterval: int) : TrainingResult =

    let redTable    = createQTable ()
    let yellowTable = createQTable ()
    let history     = System.Collections.Generic.List<int * int * float>()

    let epsilonDecay = (epsilonStart - epsilonEnd) / float episodes

    let mutable redWins   = 0
    let mutable totalGames = 0

    for ep in 1 .. episodes do
        let epsilon = max epsilonEnd (epsilonStart - epsilonDecay * float ep)
        let outcome = playEpisode rng redTable yellowTable alpha gamma epsilon
        totalGames <- totalGames + 1
        match outcome with
        | RedWon -> redWins <- redWins + 1
        | _ -> ()

        if ep % logInterval = 0 then
            let redWinRate = float redWins / float totalGames
            history.Add(ep, redTable.Count, redWinRate)

    {
        RedTable      = redTable
        YellowTable   = yellowTable
        TotalEpisodes = episodes
        History       = history |> Seq.toList
    }

/// Play a single game between a Q-agent (using its trained table, epsilon=0) and a random opponent.
/// agentPlayer: which color the Q-agent plays (Red or Yellow).
/// Returns: winner cell option (None on draw), and number of moves played.
let playQAgentVsRandom
    (rng: System.Random)
    (table: QTable)
    (agentPlayer: Cell)
    : Cell option * int =

    let rec loop board currentPlayer moves =
        match isGameOver board with
        | Some RedWins    -> Some Red, moves
        | Some YellowWins -> Some Yellow, moves
        | Some Draw       -> None, moves
        | None ->
            let col =
                if currentPlayer = agentPlayer then
                    chooseAction rng table board 0.0  // greedy (epsilon=0)
                else
                    let legal = legalMoves board
                    legal.[rng.Next(legal.Length)]
            loop (applyMove board currentPlayer col) (opponent currentPlayer) (moves + 1)

    loop (emptyBoard ()) Red 0
```
