Phase 2의 한계: 상태 공간 폭발

2장에서 우리는 틱택토의 가치 함수를 Map<Board, float>로 표현했다. 도달 가능한 상태가 ~5,478개였기 때문에 이 방식이 완벽히 작동했다.

커넥트 포는 본질적으로 다르다. 6×7 보드에서 가능한 위치 수는 천문학적이다.

커넥트 포 게임 규칙

커넥트 포는 두 플레이어가 교대로 6행 7열 격자에 돌을 떨어뜨리는 게임이다. 돌은 중력을 받아 열의 최하단 빈 칸에 쌓인다. 가로, 세로, 대각선 방향으로 4개를 먼저 연결하면 승리한다.

핵심 F# 타입 정의

module ConnectFour.Domain

type Cell = Empty | Red | Yellow

/// 6×7 board as flat 42-element array, row-major: index = row * 7 + col
/// Row 0 = top, Row 5 = bottom
type Board = Cell array

type GameState = {
    Board: Board
    CurrentPlayer: Cell
}

let rows = 6
let cols = 7

let inline idx row col = row * cols + col

let emptyBoard () : Board = Array.create (rows * cols) Empty

let initialState () : GameState =
    { Board = emptyBoard (); CurrentPlayer = Red }

let opponent (player: Cell) : Cell =
    match player with
    | Red    -> Yellow
    | Yellow -> Red
    | Empty  -> failwith "opponent: Empty is not a player"

보드는 42개 원소의 1차원 배열이다. idx row col = row * 7 + col로 접근한다. 행 0이 맨 위, 행 5가 맨 아래(중력 기준점)다.

게임 규칙 엔진

module ConnectFour.Rules

open ConnectFour.Domain

let dropRow (board: Board) (col: int) : int option =
    [ rows - 1 .. -1 .. 0 ]
    |> List.tryFind (fun row -> board.[idx row col] = Empty)

let legalMoves (board: Board) : int list =
    [ 0 .. cols - 1 ]
    |> List.filter (fun col -> dropRow board col |> Option.isSome)

let applyMove (board: Board) (player: Cell) (col: int) : Board =
    match dropRow board col with
    | None -> failwith $"applyMove: column {col} is full"
    | Some row ->
        board |> Array.mapi (fun i c -> if i = idx row col then player else c)

let private checkDirection (board: Board) (r: int) (c: int) (dr: int) (dc: int) (player: Cell) : bool =
    [ 0 .. 3 ]
    |> List.forall (fun k ->
        let nr = r + k * dr
        let nc = c + k * dc
        nr >= 0 && nr < rows && nc >= 0 && nc < cols && board.[idx nr nc] = player)

let checkWinner (board: Board) : Cell option =
    let directions = [ (0,1); (1,0); (1,1); (1,-1) ]
    [ for r in 0 .. rows-1 do
      for c in 0 .. cols-1 do
      for (dr, dc) in directions do
        yield (r, c, dr, dc) ]
    |> List.tryPick (fun (r, c, dr, dc) ->
        let cell = board.[idx r c]
        if cell <> Empty && checkDirection board r c dr dc cell
        then Some cell
        else None)

type GameResult = RedWins | YellowWins | Draw

let isGameOver (board: Board) : GameResult option =
    match checkWinner board with
    | Some Red    -> Some RedWins
    | Some Yellow -> Some YellowWins
    | Some Empty  -> None
    | None ->
        if Array.forall (fun c -> c <> Empty) board then Some Draw
        else None

dropRow는 열에서 돌이 실제로 떨어질 위치를 찾는다 — 아래에서부터 첫 번째 빈 칸이다. isGameOver는 승자 확인(4방향)과 무승부 확인(보드가 모두 찼을 때)을 처리한다.

Minimax Alpha-Beta Pruning

Minimax는 상대방도 최선을 다한다고 가정하고 게임 트리를 완전 탐색한다. 나는 내 점수를 최대화하려 하고, 상대는 내 점수를 최소화하려 한다.

Alpha-Beta Pruning은 Minimax를 최적화한다. 이미 더 좋은 수를 알고 있는 경우, 탐색할 필요 없는 트리 가지를 잘라낸다. 평균적으로 탐색 비용을 절반으로 줄인다.

Negamax 변형을 사용하면 두 플레이어를 동일한 코드로 처리할 수 있다: 모든 상태의 점수를 “현재 플레이어 기준“으로 평가한다.

module ConnectFour.Minimax

open ConnectFour.Domain
open ConnectFour.Rules

let [<Literal>] private NegInf = -1_000_000
let [<Literal>] private PosInf =  1_000_000

let private scoreWindow (window: Cell list) (player: Cell) : int =
    let opp = opponent player
    let p = window |> List.filter ((=) player) |> List.length
    let o = window |> List.filter ((=) opp) |> List.length
    if o > 0 then 0
    else
        match p with
        | 4 -> 10000
        | 3 -> 50
        | 2 -> 3
        | 1 -> 1
        | _ -> 0

let private allWindows (board: Board) : Cell list list =
    [ for r in 0 .. rows-1 do
        for c in 0 .. cols-4 do
          yield [ board.[idx r c]; board.[idx r (c+1)]; board.[idx r (c+2)]; board.[idx r (c+3)] ]
      for c in 0 .. cols-1 do
        for r in 0 .. rows-4 do
          yield [ board.[idx r c]; board.[idx (r+1) c]; board.[idx (r+2) c]; board.[idx (r+3) c] ]
      for r in 0 .. rows-4 do
        for c in 0 .. cols-4 do
          yield [ board.[idx r c]; board.[idx (r+1) (c+1)]; board.[idx (r+2) (c+2)]; board.[idx (r+3) (c+3)] ]
      for r in 3 .. rows-1 do
        for c in 0 .. cols-4 do
          yield [ board.[idx r c]; board.[idx (r-1) (c+1)]; board.[idx (r-2) (c+2)]; board.[idx (r-3) (c+3)] ]
    ]

let evaluateBoard (board: Board) (player: Cell) : int =
    let opp = opponent player
    let windows = allWindows board
    let netWindowScore =
        windows |> List.sumBy (fun w ->
            scoreWindow w player - scoreWindow w opp)
    let centerBonus =
        [ 0 .. rows-1 ]
        |> List.filter (fun r -> board.[idx r 3] = player)
        |> List.length
        |> (*) 3
    netWindowScore + centerBonus

let rec minimaxAB
    (board: Board) (player: Cell) (depth: int)
    (alpha: int) (beta: int) (pruneCount: int ref) : int =
    match isGameOver board with
    | Some result ->
        match result with
        | RedWins    -> if player = Red    then 10000 + depth else -(10000 + depth)
        | YellowWins -> if player = Yellow then 10000 + depth else -(10000 + depth)
        | Draw       -> 0
    | None when depth = 0 ->
        evaluateBoard board player
    | None ->
        let opp = opponent player
        let moves = legalMoves board |> List.sortBy (fun c -> abs (c - 3))
        let rec loop moves alpha bestScore =
            match moves with
            | [] -> bestScore
            | col :: rest ->
                let nextBoard = applyMove board player col
                let childScore = -(minimaxAB nextBoard opp (depth - 1) (-beta) (-alpha) pruneCount)
                let newBest = max bestScore childScore
                let newAlpha = max alpha newBest
                if newAlpha >= beta then
                    pruneCount.Value <- pruneCount.Value + 1
                    newBest
                else
                    loop rest newAlpha newBest
        loop moves alpha NegInf

let chooseMoveAB (board: Board) (player: Cell) (depth: int) : int * int =
    let pruneCount = ref 0
    let opp = opponent player
    let moves = legalMoves board |> List.sortBy (fun c -> abs (c - 3))
    let scored =
        moves |> List.map (fun col ->
            let nextBoard = applyMove board player col
            let score = -(minimaxAB nextBoard opp (depth - 1) NegInf PosInf pruneCount)
            col, score)
    let bestCol = scored |> List.maxBy snd |> fst
    bestCol, pruneCount.Value

let rec private naiveMinimax (board: Board) (player: Cell) (depth: int) : int =
    match isGameOver board with
    | Some result ->
        match result with
        | RedWins    -> if player = Red    then 10000 + depth else -(10000 + depth)
        | YellowWins -> if player = Yellow then 10000 + depth else -(10000 + depth)
        | Draw       -> 0
    | None when depth = 0 ->
        evaluateBoard board player
    | None ->
        let opp = opponent player
        let moves = legalMoves board |> List.sortBy (fun c -> abs (c - 3))
        moves |> List.map (fun col ->
            let nextBoard = applyMove board player col
            -(naiveMinimax nextBoard opp (depth - 1)))
        |> List.max

let chooseMoveNaive (board: Board) (player: Cell) (depth: int) : int =
    let opp = opponent player
    let moves = legalMoves board |> List.sortBy (fun c -> abs (c - 3))
    moves |> List.map (fun col ->
        let nextBoard = applyMove board player col
        col, -(naiveMinimax nextBoard opp (depth - 1)))
    |> List.maxBy snd |> fst

chooseMoveAB는 (최선 열, 누적 가지치기 횟수) 쌍을 반환한다. 가지치기 횟수는 Alpha-Beta가 얼마나 효과적으로 탐색을 줄였는지 보여주는 지표다.

Q-Learning

Q-Learning은 행동-가치 함수 Q(s, a)를 학습한다: “상태 s에서 행동 a를 취할 때 기대 보상은 얼마인가?”

벨만 방정식으로 업데이트한다:

Q(s,a) ← Q(s,a) + α × [r + γ × max_a' Q(s',a') - Q(s,a)]
  • α (alpha): 학습률 — 새 정보를 얼마나 빠르게 반영하나
  • γ (gamma): 할인율 — 미래 보상을 얼마나 중요하게 여기나
  • r + γ × max Q(s',a'): TD 목표값
  • r + γ × max Q(s',a') - Q(s,a): TD 오차

ε-greedy 정책으로 탐색과 활용의 균형을 맞춘다: ε 확률로 무작위 행동(탐색), 1-ε 확률로 최선 행동(활용).

module ConnectFour.QAgent

open ConnectFour.Domain
open ConnectFour.Rules

type QTable = System.Collections.Generic.Dictionary<string, float[]>

let encodeState (board: Board) : string =
    board
    |> Array.map (fun c ->
        match c with
        | Red    -> 'R'
        | Yellow -> 'Y'
        | Empty  -> '.')
    |> System.String

let createQTable () : QTable = QTable()

let getQ (table: QTable) (state: string) : float[] =
    match table.TryGetValue(state) with
    | true, values -> values
    | false, _ ->
        let values = Array.create cols 0.0
        table.[state] <- values
        values

let chooseAction (rng: System.Random) (table: QTable) (board: Board) (epsilon: float) : int =
    let legal = legalMoves board
    if legal.IsEmpty then failwith "chooseAction: no legal moves (game should be over)"
    if rng.NextDouble() < epsilon then
        legal.[rng.Next(legal.Length)]
    else
        let state = encodeState board
        let qVals = getQ table state
        legal |> List.maxBy (fun c -> qVals.[c])

let updateQ
    (table: QTable) (state: string) (action: int) (reward: float)
    (nextState: string) (nextLegalMoves: int list)
    (alpha: float) (gamma: float) (isTerminal: bool) : unit =
    let qCurr = getQ table state
    let nextMax =
        if isTerminal || nextLegalMoves.IsEmpty then 0.0
        else
            let qNext = getQ table nextState
            nextLegalMoves |> List.map (fun c -> qNext.[c]) |> List.max
    let target = reward + gamma * nextMax
    qCurr.[action] <- qCurr.[action] + alpha * (target - qCurr.[action])

let [<Literal>] RewardWin  =  1.0
let [<Literal>] RewardLoss = -1.0
let [<Literal>] RewardDraw =  0.3
let [<Literal>] RewardStep =  0.0

Q-Table은 Dictionary<string, float[]>로 구현된다:

  • 키: 42글자 문자열 (. = 빈칸, R = 빨강, Y = 노랑)
  • 값: 7개 원소 배열 (각 열의 Q값)

Red와 Yellow가 별도 테이블을 유지한다 — 각자 자신의 관점에서 학습한다.

Q-Table의 한계 — 왜 Phase 4 DQN이 필요한가

50,000 에피소드 학습 후의 현실:

=== Q-Table 크기 분석 ===
학습 후 방문한 상태: 169,311
전체 가능한 상태:    4,531,985,219,092
커버율:              0.000004%

Q-Table은 방문한 상태만 저장한다. 4조 개 상태 중 0.000004%만 본 것이다. 한 번도 보지 못한 상태에서는 Q값이 0으로 초기화되어 있다 — 에이전트가 판단 근거를 갖지 못한다.

이 방식은 일반화가 불가능하다. “이 위치와 비슷한 위치에서 이겼었다“는 지식을 전달할 수 없다.

AI vs AI 결과가 이를 극명하게 보여준다:

=== AI vs AI 결과 (총 20게임) ===
Minimax(R) 승: 20 / 20 (100.0%)
Q-Learning(Y) 승: 0 / 20 (0.0%)
Alpha-Beta 누적 가지치기: 184,820회

Minimax는 매 수마다 게임 트리를 깊이 6까지 탐색하며 최선 수를 계산한다. Q-Learning은 본 적 없는 보드 배치에서 갈피를 잡지 못한다.

Phase 4 미리보기: 신경망으로 일반화

Q-Table의 한계를 극복하는 방법은 **함수 근사(Function Approximation)**다.

상태를 키-값 쌍으로 저장하는 대신, 신경망으로 Q(s,a)를 근사한다:

  • 입력: 보드 배치 (42개 픽셀로 본다)
  • 출력: 7개의 Q값 (각 열에 대해)

이것이 **DQN(Deep Q-Network)**이다. 2015년 DeepMind가 Atari 게임을 정복하며 세상을 놀라게 한 방법.

신경망은 비슷한 보드 배치에서 비슷한 Q값을 예측할 수 있다 — 일반화가 가능하다. 4조 개 상태를 모두 방문하지 않아도 된다. 패턴을 학습하면 된다.

4장에서는 TorchSharp로 DQN을 F#으로 구현하고, 같은 커넥트 포 환경에서 Q-Table이 실패하는 곳에서 DQN이 성공하는지 확인한다.
